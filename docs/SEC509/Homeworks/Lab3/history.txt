   66  function get_client_ip() {
   67      if (!empty($_SERVER['HTTP_CLIENT_IP'])) {
   68          $ip = $_SERVER['HTTP_CLIENT_IP'];
   69      } elseif (!empty($_SERVER['HTTP_X_FORWARDED_FOR'])) {
   70          $ip = explode(',', $_SERVER['HTTP_X_FORWARDED_FOR'])[0];
   71      } else {
   72          $ip = $_SERVER['REMOTE_ADDR'];
   73      }
